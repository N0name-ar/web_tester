import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const BASE_TEMP_MAIL = "https://api.internal.temp-mail.io/api/v3"; const TEMP_HEADERS = { "Content-Type": "application/json; charset=UTF-8", "User-Agent": "okhttp/4.12.0", "Host": "api.internal.temp-mail.io", }; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type, x-vip-signature, x-copyright-check", }; Deno.serve(async (req) => { // Handle Preflight Request CORS if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { // 🔑 CHECK HEADER KEAMANAN const KUNCI_HEADER_AMAN = req.headers.get("X-Copyright-Check"); if (KUNCI_HEADER_AMAN !== "titik-temu-cloud-secured-v25-key-auth") { return new Response( JSON.stringify({ success: false, error: "❌ Akses Ilegal!" }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 403 } ); } // 🔑 INITIALIZE SUPABASE CLIENT const supabaseUrl = Deno.env.get("SUPABASE_URL")!; const supabaseServiceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const supabase = createClient(supabaseUrl, supabaseServiceKey); // Query ke tabel vault_sultan const { data: vaultData, error: vaultError } = await supabase .from("vault_sultan") .select("SULTAN_MAIN") .eq("id", 1) .single(); if (vaultError || !vaultData || !vaultData.SULTAN_MAIN) { return new Response( JSON.stringify({ success: false, error: `CRITICAL: Gagal membaca tabel 'vault_sultan'! Detail: ${vaultError?.message || "Data kosong"}`, }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 500 } ); } const MASTER_COMPRESDATA_BASE64 = vaultData.SULTAN_MAIN; // Parse Payload Input const { custom_name, password, silver, gold, exp, vip_key } = await req.json(); if (!custom_name || !password) { return new Response( JSON.stringify({ success: false, error: "Nama Depan dan Password wajib diisi, Wak!" }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 400 } ); } // ================= 2. GENERATE DISPOSABLE TEMP-MAIL ================= const domainRes = await fetch(`${BASE_TEMP_MAIL}/domains`, { headers: TEMP_HEADERS }); if (!domainRes.ok) throw new Error("Gagal mengambil active domains dari Temp-Mail API."); const domainData = await domainRes.json(); const domains = domainData.domains || []; if (domains.length === 0) throw new Error("Tidak ada domain publik yang tersedia di Temp-Mail."); const randomDomain = domains[Math.floor(Math.random() * domains.length)].name; const randomBack = Math.random().toString(36).substring(2, 8); const cleanCustomName = custom_name.replace(/[^a-zA-Z0-9]/g, ""); const username = `${cleanCustomName}_${randomBack}`; const emailAddr = `${username}@${randomDomain}`; const createEmailRes = await fetch(`${BASE_TEMP_MAIL}/email/new`, { method: "POST", headers: TEMP_HEADERS, body: JSON.stringify({ domain: randomDomain, name: username }), }); if (!createEmailRes.ok) throw new Error("Gagal mengalokasikan mailbox baru di Temp-Mail."); // ================= 3. REGISTER HANDSHAKE KE CARX SERVER ================= const devId = crypto.randomUUID(); const uId = crypto.randomUUID().replace(/-/g, "").substring(0, 32); const regUrl = "https://carx-id-prod.carx-online.com/api/auth/register"; const regBody = new URLSearchParams({ deviceId: devId, deviceUniqueId: uId, username: emailAddr, password: password, project: "STREET", }); const resReg = await fetch(regUrl, { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: regBody, }); const regJson = await resReg.json(); const token = regJson?.d?.token || regJson?.token; const realNick = regJson?.d?.carxId || regJson?.d?.username || username; if (!token) { throw new Error(`Register CarX Gagal Ditolak: ${JSON.stringify(regJson)}`); } // ================= 4. AUTOMATED OTP POLLING INTERCEPTOR ================= let otpCode: string | null = null; const startTime = Date.now(); const timeoutMs = 120000; // Timeout Max 2 Menit while (Date.now() - startTime < timeoutMs) { await new Promise((r) => setTimeout(r, 4000)); // Sleep 4 Detik tiap polling const msgRes = await fetch(`${BASE_TEMP_MAIL}/email/${emailAddr}/messages`, { headers: TEMP_HEADERS }); if (msgRes.ok) { const messages = await msgRes.json(); if (messages && messages.length > 0) { for (const msg of messages) { const rawContent = `${msg.body || ""} ${msg.body_html || ""}`; const cleanText = rawContent.replace(/<[^>]*>/g, " ").replace(/\s+/g, " "); // Multi-Tier Regex Matcher const match1 = cleanText.match(/Your unique confirmation code[\s:]*([a-zA-Z0-9]{4,8})/i); const match2 = cleanText.match(/(?:confirmation code|code|otp)[\s:]*([a-zA-Z0-9]{4,8})/i); const match3 = cleanText.match(/\b([a-f0-9]{6})\b/i); otpCode = match1?.[1] || match2?.[1] || match3?.[1] || null; if (otpCode) break; } } } if (otpCode) break; } if (!otpCode) throw new Error("OTP Timeout: Kode verifikasi tidak ditemukan di Inbox Temp-Mail."); // ================= 5. VERIFY AKUN CARX ================= const verifyUrl = "https://carx-id-prod.carx-online.com/api/auth/verify"; const resVerify = await fetch(verifyUrl, { method: "POST", headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/x-www-form-urlencoded", "User-Agent": "UnityPlayer/6000.0.64f1 (UnityWebRequest/1.0, libcurl/8.10.1-DEV)", "X-Unity-Version": "6000.0.64f1", }, body: new URLSearchParams({ code: otpCode }), }); if (!resVerify.ok) throw new Error("Verifikasi OTP ke CarX Auth Server gagal."); // ================= 6. UNCOMPRESS GZIP, PATCH JSON & RE-COMPRESS ================= const cleanVaultBase64 = MASTER_COMPRESDATA_BASE64.trim().replace(/\s/g, ""); const rawBin = Uint8Array.from(atob(cleanVaultBase64), (c) => c.charCodeAt(0)); // Deteksi Magic Bytes Marker Header GZIP (0x1f, 0x8b) let gzStart = -1; for (let i = 0; i < rawBin.length - 1; i++) { if (rawBin[i] === 0x1f && rawBin[i + 1] === 0x8b) { gzStart = i; break; } } if (gzStart === -1) { throw new Error("Format master biner di tabel 'vault_sultan' invalid: Magic Bytes GZIP (0x1f8b) tidak ditemukan!"); } const headerAsli = rawBin.slice(0, gzStart); const gzBody = rawBin.slice(gzStart); // Decompress Body Payload const decompressedStream = new Response(gzBody).body!.pipeThrough(new DecompressionStream("gzip")); const profileJsonText = await new Response(decompressedStream).text(); const profile = JSON.parse(profileJsonText); // Modifikasi Resource Profile profile.nickname = realNick; profile.resources = profile.resources || {}; profile.resources.soft = { amount: Number(silver || 9999999) }; profile.resources.hard = { amount: Number(gold || 999999) }; profile.resources.experience = { amount: Number(exp || 999999) }; profile.last_save_time = Math.floor(Date.now() / 1000); // Re-Compress Body Payload ke GZIP const jsonBytes = new TextEncoder().encode(JSON.stringify(profile)); const compressedStream = new Response(jsonBytes).body!.pipeThrough(new CompressionStream("gzip")); const newGzBuffer = await new Response(compressedStream).arrayBuffer(); const newGzArray = new Uint8Array(newGzBuffer); // Re-combine Header Asli + New GZIP Payload const finalPayloadArray = new Uint8Array(headerAsli.length + newGzArray.length); finalPayloadArray.set(headerAsli, 0); finalPayloadArray.set(newGzArray, headerAsli.length); // Encode Biner Akhir ke Base64 String let binaryStr = ""; for (let i = 0; i < finalPayloadArray.length; i++) { binaryStr += String.fromCharCode(finalPayloadArray[i]); } const finalPayloadBase64 = btoa(binaryStr); // ================= 7. INJECT SAVE DATA KE CARX PROFILE SERVER ================= const resInject = await fetch("https://street-prod.carx-online.com/str/v1/client/profiles", { method: "POST", headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json", "User-Agent": "UnityPlayer/6000.0.64f1 (UnityWebRequest/1.0, libcurl/8.10.1-DEV)", }, body: JSON.stringify({ compressed_data: finalPayloadBase64 }), }); if (!resInject.ok) { throw new Error(`Profile Injection Rejected: ${await resInject.text()}`); } // ================= 8. RETURN RESPONSE KE CLIENT ================= return new Response( JSON.stringify({ success: true, data: { email: emailAddr, password: password, username: realNick, silver: Number(silver || 9999999), gold: Number(gold || 999999), exp: Number(exp || 999999), token: token, otp: otpCode, }, }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 200 } ); } catch (err: any) { return new Response( JSON.stringify({ success: false, error: err.message || "Internal Cloud Execution Error" }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 500 } ); } });
